# ================= Helper Functions =================
def detect_stream_type(url):
    url_lower = url.lower()
    if any(x in url_lower for x in ['amagi', 'amg', 'now3', 'playout']):
        return "amagi"
    elif url_lower.endswith('.m3u8'):
        return "hls"
    elif url_lower.endswith(('.mp4', '.ts', '.mkv', '.webm')):
        return "direct_video"
    elif 'youtube.com' in url_lower or 'youtu.be' in url_lower:
        return "youtube"
    elif 'twitch.tv' in url_lower:
        return "twitch"
    else:
        return "direct"

def get_headers_for_stream(url):
    stream_type = detect_stream_type(url)
    if stream_type == "amagi":
        return get_amagi_headers(url)
    else:
        return BROWSER_HEADERS

def get_owner_link():
    if OWNER_USERNAME:

def quality_keyboard(chat_id, channel_name):
    """Generate inline keyboard for quality selection"""
    buttons = []
    row = []
    for i, q in enumerate(QUALITY_PRESETS.keys()):
        row.append(InlineKeyboardButton(q, callback_data=f"quality_{chat_id}_{channel_name}_{q}"))
        if len(row) == 3:
            buttons.append(row)
            row = []
    if row:
        buttons.append(row)
    buttons.append([InlineKeyboardButton("⏹ Stop Stream", callback_data=f"stop_{chat_id}")])
    return InlineKeyboardMarkup(buttons)

async def play_stream(chat_id, channel_name, quality, message_obj=None, is_movie=False, series_info=None):
    """Universal stream player for TV, movies, series episodes"""
    if is_movie:
        stream_url = series_info["url"] if series_info else CHANNELS.get(channel_name)
        title = channel_name
    elif series_info:
        stream_url = series_info["url"]
        title = f"{series_info['series_name']} S{series_info['season']}E{series_info['episode']} - {series_info['episode_title']}"
    else:
        stream_url = CHANNELS.get(channel_name)
        title = channel_name.title()

    if not stream_url:
        if message_obj:
            await message_obj.reply("❌ Stream URL not found.")
        return

    stream_type = detect_stream_type(stream_url)
    quality_label = quality.upper().replace("P", "p")
    video_quality = QUALITY_PRESETS.get(quality, VideoQuality.HD_720p)
    headers = get_headers_for_stream(stream_url)

    msg = await message_obj.reply(
        f"🎬 **Starting Stream...**\n"
        f"📺 **{title}**\n"
        f"📊 **{quality_label}**\n"
        f"🔧 **Type:** {stream_type.upper()}\n"
        f"⏳ Please wait 5-10 seconds..."
    )

    try:
        await call_py.play(
            chat_id,
            MediaStream(
                media_path=stream_url,
                video_parameters=video_quality,
                headers=headers
            )
        )
        active_chats.add(chat_id)
        current_streams[chat_id] = {
            "channel": channel_name,
            "quality": quality,
            "stream_type": stream_type,
            "title": title,
            "url": stream_url,
            "is_movie": is_movie,
            "series_info": series_info
        }

        await msg.edit_text(
            f"▶️ **Now Streaming!**\n\n"
            f"📺 **{title}**\n"
            f"📊 **{quality_label}**\n"
            f"🔧 **Type:** {stream_type.upper()}\n"
            f"📡 **Status:** Live\n\n"
            f"⚙️ **Change quality** using buttons below.\n"
            f"💡 `/stopvc` to stop.",
            reply_markup=quality_keyboard(chat_id, channel_name)
        )

        # Notify owner
        try:
            await app.send_message(OWNER_ID, f"🟢 Stream ON: {title} ({quality_label})\nChat: {chat_id}")
        except:
            pass

    except Exception as e:
        error_text = str(e)[:200]
        await msg.edit_text(
            f"❌ **Failed to Play Stream**\n\n"
            f"📺 **{title}**\n"
            f"⚠️ **Error:** `{error_text}`\n\n"
            f"**Try:**\n"
            f"• Lower quality: `/hqlivetv {channel_name} 360p`\n"
            f"• Wait a few seconds and try again"
        )

async def change_stream_quality(chat_id, new_quality):
    """Change quality of currently playing stream"""
    if chat_id not in current_streams:
        return False, "No active stream in this chat."
    stream_info = current_streams[chat_id]
    try:
        await call_py.change_stream(
            chat_id,
            MediaStream(
                media_path=stream_info["url"],
                video_parameters=QUALITY_PRESETS.get(new_quality, VideoQuality.HD_720p),
                headers=get_headers_for_stream(stream_info["url"])
            )
        )
        stream_info["quality"] = new_quality
        current_streams[chat_id] = stream_info
        return True, f"Quality changed to {new_quality.upper()}"
    except Exception as e:
        return False, str(e)[:100]

async def ensure_assistant_in_group(chat_id):
    """Check if user account is in group; if not, try to join via invite link"""
    try:
        member = await user_app.get_chat_member(chat_id, (await user_app.get_me()).id)
        if member.status in ["member", "administrator", "creator"]:
            return True
    except:
        pass
    # Not a member, try to join
    try:
        # Bot creates an invite link (requires admin rights)
        invite_link = await app.export_chat_invite_link(chat_id)
        await user_app.join_chat(invite_link)
        return True
    except Exception as e:
        print(f"Failed to add assistant to {chat_id}: {e}")
        return False

    async def wrapper(client, message):
        if message.chat.type in ["group", "supergroup"]:
            if message.from_user and message.from_user.id == OWNER_ID:
                return await func(client, message)
            approved = load_approved_groups()
            if message.chat.id not in approved:
                await message.reply(
                    "🚫 **Group not approved**\n\n"
                    "This bot is restricted to approved groups only.\n"
                    "Contact the owner to request approval."
                )
                return
        return await func(client, message)
    return wrapper

@app.on_message(filters.command("playepisode"))
@group_approved
async def play_episode(client, message):
    if len(message.command) < 2:
        return await message.reply("**Usage:** `/playepisode SeriesName S01E01`")
    try:
        parts = message.text.split(None, 1)[1].split()
        if len(parts) != 2:
            raise ValueError
        series_name, se = parts[0], parts[1].upper()
        # Parse S01E01 format
        match = re.match(r'S(\d+)E(\d+)', se)
        if not match:
            return await message.reply("❌ Invalid format. Use `S01E01`")
        season = int(match.group(1))
        episode = int(match.group(2))
        episodes = get_episodes(series_name)
        target = None
        for s, e, title, url in episodes:
            if s == season and e == episode:
                target = {"series_name": series_name, "season": s, "episode": e, "episode_title": title, "url": url}
                break
        if not target:
            return await message.reply(f"❌ Episode S{season}E{episode} not found in {series_name}")
        await play_stream(message.chat.id, f"{series_name} S{season}E{episode}", "720p", message, is_movie=False, series_info=target)
    except:
        await message.reply("**Usage:** `/playepisode SeriesName S01E01`")

@app.on_message(filters.command("addxtream") & filters.user(OWNER_ID))
async def add_xtream(client, message):
    args = message.text.split()
    if len(args) != 4:
        return await message.reply("**Usage:** `/addxtream <URL> <Username> <Password>`")
    url, user, passwd = args[1].rstrip("/"), args[2], args[3]
    api_url = f"{url}/player_api.php?username={user}&password={passwd}&action=get_live_streams"
    msg = await message.reply("🔄 Fetching channels...")
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(api_url, timeout=30) as resp:
                if resp.status == 200:
                    data = await resp.json()
                    count = 0
                    for stream in data:
                        name = stream.get("name", "").strip().lower()
                        if name:
                            stream_url = f"{url}/live/{user}/{passwd}/{stream.get('stream_id')}.m3u8"
                            CHANNELS[name] = stream_url
                            save_channel_to_db(name, stream_url)
                            count += 1
                    await msg.edit_text(f"✅ Loaded {count} channels!")
                else:
                    await msg.edit_text(f"❌ Status {resp.status}")
    except Exception as e:
        await msg.edit_text(f"❌ `{str(e)[:200]}`")

# ================= Callback Query Handler =================
@app.on_callback_query()
async def callback_handler(client, callback_query: CallbackQuery):
    data = callback_query.data
    chat_id = callback_query.message.chat.id
    user_id = callback_query.from_user.id

    if data.startswith("quality_"):
        # Format: quality_chatId_channelName_quality
        parts = data.split("_")
        if len(parts) >= 5:
            try:
                target_chat_id = int(parts[1])
                channel_name = "_".join(parts[2:-1])  # channel name may contain underscores
                new_quality = parts[-1]
                if target_chat_id != chat_id:
                    await callback_query.answer("This button is for another chat.", show_alert=True)
                    return
                # Check if user is admin or owner? For simplicity, allow any user to change quality
                success, msg = await change_stream_quality(chat_id, new_quality)
                if success:
                    await callback_query.answer(f"Quality changed to {new_quality.upper()}", show_alert=False)
                    await callback_query.message.edit_reply_markup(reply_markup=quality_keyboard(chat_id, channel_name))
                else:
                    await callback_query.answer(f"Failed: {msg}", show_alert=True)
            except Exception as e:
                await callback_query.answer(f"Error: {str(e)[:50]}", show_alert=True)

    elif data.startswith("stop_"):
        target_chat_id = int(data.split("_")[1])
        if target_chat_id != chat_id:
            await callback_query.answer("This button is for another chat.", show_alert=True)
            return
        try:
            await call_py.leave_call(chat_id)
            active_chats.discard(chat_id)
            current_streams.pop(chat_id, None)
            await callback_query.message.edit_text("⏹ **Stream stopped by user.**")
            await callback_query.answer("Stream stopped.", show_alert=False)
        except Exception as e:
            await callback_query.answer(f"Error: {str(e)[:50]}", show_alert=True)
