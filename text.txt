def get_amagi_headers(url):
    parsed = urlparse(url)
    base_url = f"{parsed.scheme}://{parsed.netloc}"
    return {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
        "Accept": "*/*",
        "Origin": base_url,
        "Referer": base_url + "/",

def detect_stream_type(url):
    url_lower = url.lower()
    if any(x in url_lower for x in ['amagi', 'amg', 'now3', 'playout']):
        return "amagi"
    elif url_lower.endswith('.m3u8') or '.m3u' in url_lower:
        return "hls"
    elif url_lower.endswith(('.mp4', '.ts', '.mkv', '.webm')):
        return "direct_video"
    else:
        return "direct"

def get_headers_for_stream(url):
    if detect_stream_type(url) == "amagi":
        return get_amagi_headers(url)
    return BROWSER_HEADERS

def check_approval(func):
    """Decorator to block unauthorized groups"""
    async def wrapper(client, message):
        chat_id = message.chat.id
        if chat_id != OWNER_ID and chat_id not in DB["approved_groups"] and message.chat.type in ["group", "supergroup"]:
            return await message.reply("🚫 **[𝗔𝗖𝗖𝗘𝗦𝗦 𝗗𝗘𝗡𝗜𝗘𝗗]**\n`This group is not authorized to use the bot. Ask the owner to /approve this group.`")
        
        # Track served chats
        if chat_id not in DB["served_chats"]:
            DB["served_chats"].append(chat_id)
            save_data()
            
        return await func(client, message)
    return wrapper

@app.on_message(filters.command(["livetv", "movie", "series"]))
@check_approval
async def play_media(client, message):
    chat_id = message.chat.id
    cmd = message.command[0].lower()
    
    if len(message.command) < 2:
        return await message.reply(f"⚠️ **[𝗦𝗬𝗡𝗧𝗔𝗫]** `/{cmd} <Name>`")
    
    name = " ".join(message.command[1:]).strip().lower()
    
    category_map = {"livetv": "channels", "movie": "movies", "series": "series"}
    db_key = category_map[cmd]
    
    if name not in DB[db_key]:
        return await message.reply(f"❌ **[𝗡𝗢𝗧 𝗙𝗢𝗨𝗡𝗗]** `That {cmd} doesn't exist.`")
    
    stream_url = DB[db_key][name]
    
    # 1. Ensure Assistant is in VC
    if not await ensure_assistant(chat_id, message):
        return
    
    # 2. Start Playback
    msg = await message.reply("🔄 **[𝗦𝗬𝗦𝗧𝗘𝗠]** `Connecting to global servers...`")
    
    try:
        quality = "720p"
        headers = get_headers_for_stream(stream_url)
        
        await call_py.play(
            chat_id,
            MediaStream(media_path=stream_url, video_parameters=QUALITY_PRESETS[quality], headers=headers)
        )
        
        active_streams[chat_id] = {"name": name, "url": stream_url, "category": db_key, "msg_id": msg.id}
        
        await msg.edit_text(
            format_stream_message(name, cmd, quality),
            reply_markup=get_quality_keyboard()
        )
        
    except Exception as e:
        await msg.edit_text(f"❌ **[𝗦𝗧𝗥𝗘𝗔𝗠 𝗘𝗥𝗥𝗢𝗥]**\n`{str(e)[:200]}`")

# ================= 🎛 INLINE BUTTONS (QUALITY CHANGE) 🎛 =================
@app.on_callback_query(filters.regex(r"^(q:|stop_stream)"))
async def callback_handler(client, callback_query: CallbackQuery):
    chat_id = callback_query.message.chat.id
    data = callback_query.data
    
    if data == "stop_stream":
        try:
            await call_py.leave_call(chat_id)
            if chat_id in active_streams:
                del active_streams[chat_id]
            await callback_query.message.edit_text("🛑 **[𝗦𝗬𝗦𝗧𝗘𝗠]** `Stream Terminated by User.`")
        except:
            pass
        return

    # Quality Change Protocol
    if chat_id not in active_streams:
        return await callback_query.answer("❌ Stream is no longer active.", show_alert=True)
    
    new_quality = data.split(":")[1]
    stream_info = active_streams[chat_id]
    
    await callback_query.answer(f"🔄 Switching to {new_quality}...", show_alert=False)
    
    try:
        headers = get_headers_for_stream(stream_info["url"])
        await call_py.change_stream(
            chat_id,
            MediaStream(
                media_path=stream_info["url"],
                video_parameters=QUALITY_PRESETS[new_quality],
                headers=headers
            )
        )
        
        await callback_query.message.edit_text(
            format_stream_message(stream_info["name"], stream_info["category"], new_quality),
            reply_markup=get_quality_keyboard()
        )
    except Exception as e:
        await callback_query.answer(f"Error: {str(e)[:50]}", show_alert=True)
